cmake_minimum_required(VERSION 3.20)
project(vapor_godot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(GODOT_CPP_DIR "" CACHE PATH "godot-cpp checkout matching your Godot version")
if(NOT EXISTS "${GODOT_CPP_DIR}/CMakeLists.txt")
 message(FATAL_ERROR "Set -DGODOT_CPP_DIR=/path/to/godot-cpp; dependency is not bundled.")
endif()
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
add_library(vapor_godot SHARED src/vapor_extension.cpp)
target_link_libraries(vapor_godot PRIVATE godot-cpp)
target_include_directories(vapor_godot PRIVATE ../../include)
set(VAPOR_NATIVE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../dist/native" CACHE PATH "Built Rust dynamic library directory")
find_library(VAPOR_LIB NAMES vapor vapor.dll PATHS "${VAPOR_NATIVE_DIR}" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(vapor_godot PRIVATE "${VAPOR_LIB}")
set_target_properties(vapor_godot PROPERTIES OUTPUT_NAME vapor_godot LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/project/bin" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/project/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set_target_properties(vapor_godot PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}/project/bin" LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}/project/bin")
endforeach()
if(APPLE)
 set_target_properties(vapor_godot PROPERTIES BUILD_RPATH "@loader_path")
else()
 set_target_properties(vapor_godot PROPERTIES BUILD_RPATH "$ORIGIN")
endif()
